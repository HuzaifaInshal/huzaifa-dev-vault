# getFileExtensionFromUrl

📁 `src/utils/get-file-extension-from-url.utils.ts`

Extracts the lowercase file extension (no dot) from a URL, stripping query strings and hash fragments. Returns an empty string if no extension is found.

## Examples

```ts
getFileExtensionFromUrl("https://cdn.example.com/photo.JPG?v=1"); // "jpg"
getFileExtensionFromUrl("https://cdn.example.com/doc.pdf#page=2"); // "pdf"
getFileExtensionFromUrl("https://cdn.example.com/noext");          // ""
```

## Source

```ts
export function getFileExtensionFromUrl(url: string): string {
  const cleanedUrl = url.split("?")[0].split("#")[0];
  const parts = cleanedUrl.trim().split("/");
  const filename = parts[parts.length - 1];
  if (filename && filename.includes(".")) {
    const extensionParts = filename.split(".");
    return extensionParts[extensionParts.length - 1].trim().toLowerCase() || "";
  }
  return "";
}
```
