# Browser Notification Utils

📁 `src/utils/browser-notification.utils.ts`

Two functions for requesting browser notification permission and showing native desktop notifications.

## requestNotificationPermission

Requests the browser's notification permission. Returns `true` if granted.

```ts
async function requestNotificationPermission(): Promise<boolean>
```

## showNotification

Shows a native browser notification. Requests permission first if not already granted. Clicking the notification navigates to the provided URL.

```ts
async function showNotification(title: string, body: string, url?: string): Promise<void>
```

| Param | Type | Description |
|-------|------|-------------|
| `title` | `string` | Notification title |
| `body` | `string` | Notification body text |
| `url` | `string` | Optional URL to navigate to on click |

## Usage

```ts
await showNotification(
  "New Message",
  "You have a new message from Alice",
  "/messages/123"
);
```

## Notes

- Only works client-side — guard with `typeof window !== "undefined"` in SSR contexts
- Requires HTTPS in production

<CodeBlock title="browser-notification.utils.ts" language="ts" source={`export const requestNotificationPermission = async () => {
  if (!("Notification" in window)) {
    alert("This browser does not support desktop notifications");
    return false;
  }
  if (Notification.permission === "granted") return true;
  const permission = await Notification.requestPermission();
  return permission === "granted";
};

export const showNotification = async (title: string, body: string, url?: string) => {
  if (Notification.permission !== "granted") {
    const granted = await requestNotificationPermission();
    if (!granted) return;
  }
  const notification = new Notification(title, { body, data: { url }, icon: "/logo/LogoIcon.png" });
  notification.onclick = (event) => {
    event.preventDefault();
    const targetUrl = notification.data?.url;
    if (targetUrl) window.location.assign(targetUrl);
  };
};`} />
